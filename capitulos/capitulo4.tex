\chapter{Análisis de Algoritmos}

\section{Introducción}

\begin{definicion}
La \textbf{demora temporal} de un algoritmo es una medida que estima el tiempo que tarda un algoritmo en completarse en función del tamaño de la entrada.
\end{definicion}

\begin{observacion}
El tiempo se mide en operaciones fundamentales, no en segundos. Depende de la entrada:
\begin{itemize}
  \item \textbf{Mejor caso}: tiempo mínimo.
  \item \textbf{Peor caso}: tiempo máximo.
  \item \textbf{Caso promedio}: promedio.
\end{itemize}
\end{observacion}

\section{Notación Asintótica}

\begin{definicion}
Sean $T, g: \Z^+ \to \R$.
\begin{itemize}
  \item $T(n) = O(g(n))$ si $\exists C_1 > 0$ tal que $|T(n)| \leq C_1 |g(n)|$ para casi todo $n$.
  \item $T(n) = \Omega(g(n))$ si $\exists C_2 > 0$ tal que $|T(n)| \geq C_2 |g(n)|$ para casi todo $n$.
  \item $T(n) = \Theta(g(n))$ si $T(n) = O(g(n))$ y $T(n) = \Omega(g(n))$.
\end{itemize}
\end{definicion}

\begin{ejemplo}
Para $T(n) = 35n^2 + 7n + 4$:
\[
T(n) \leq 46 n^2 \Rightarrow T(n) = O(n^2), \quad T(n) \geq 35 n^2 \Rightarrow T(n) = \Omega(n^2).
\]
Por tanto, $T(n) = \Theta(n^2)$.
\end{ejemplo}

\begin{observacion}
Si $T(n) = a_k n^k + \cdots + a_0$, entonces $T(n) = \Theta(n^k)$.
\end{observacion}

\begin{ejemplo}[Búsqueda lineal]
\begin{lstlisting}[style=python]
def busquedaLineal(lista, elemento):
    n = len(lista)
    i = 0
    while i < n and lista[i] != elemento:
        i = i + 1
    return i if i < n else print("no encontrado")
\end{lstlisting}
Peor caso: $T(n) = \Theta(n)$.
\end{ejemplo}

\begin{ejemplo}[Multiplicación de matrices]
\begin{lstlisting}[style=python]
def multiplicacion_matrices(A, B):
    n = len(A)
    C = [[0 for _ in range(n)] for _ in range(n)]
    for i in range(n):
        for j in range(n):
            for k in range(n):
                C[i][j] += A[i][k] * B[k][j]
    return C
\end{lstlisting}
$T(n) = n \cdot n^2 = n^3$, por lo que $T(n) = \Theta(n^3)$.
\end{ejemplo}

\begin{ejemplo}[Bucle anidado]
\begin{lstlisting}[style=python]
def contarIteraciones(n):
    x = 0
    for i in range(1, n + 1):
        for j in range(1, i + 1):
            x += 1
    return x
\end{lstlisting}
$T(n) = 1 + 2 + \cdots + n = \Theta(n^2)$.
\end{ejemplo}

\begin{ejemplo}[Subsets]
\begin{lstlisting}[style=python]
def generaSubsets(elements):
    n = len(elements)
    subsets = []
    for i in range(1 << n):
        subset = [elements[j] for j in range(n) if (i >> j) & 1]
        subsets.append(subset)
    return subsets
\end{lstlisting}
$T(n) = \Theta(n \cdot 2^n)$.
\end{ejemplo}

\begin{observacion}
En la práctica se buscan complejidades $\Theta(1)$, $\Theta(\log n)$, $\Theta(n)$ o $\Theta(n\log n)$. Las de tipo $\Theta(2^n)$, $\Theta(n!)$ son intratables para entradas moderadas.
\end{observacion}

\begin{ejemplo}[Bucle con división]
\begin{lstlisting}
j = n
while (j >= 1):
    for i = 1 to j: x = x + 1
    j = floor(j/2)
\end{lstlisting}
$T(n) = n + \lfloor n/2 \rfloor + \cdots \leq 2n$, luego $T(n) = \Theta(n)$.
\end{ejemplo}

\begin{observacion}
Un problema es \textbf{tratable} si admite un algoritmo polinomial. Es \textbf{intratable} si solo se conocen algoritmos exponenciales o factoriales. Es \textbf{irresoluble} si no existe algoritmo (por ejemplo, el problema de la parada).
\end{observacion}
