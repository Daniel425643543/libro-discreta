\chapter{Guía de Práctica 1: Técnicas Avanzadas de Conteo}

\section{Objetivos}
\begin{itemize}
  \item Aplicar los principios básicos de combinatoria.
  \item Establecer y resolver relaciones de recurrencia.
  \item Utilizar funciones generatrices.
  \item Analizar la complejidad de algoritmos.
\end{itemize}

\section{Principios Básicos de Combinatoria}

\begin{ejercicio}
¿Cuántos caracteres Braille distintos puede haber?
\end{ejercicio}

\begin{ejercicio}
¿Cuántas placas de automóvil con 3 letras seguidas de 2 dígitos?
\end{ejercicio}

\begin{ejercicio}
Manos de póquer de 5 cartas: cuatro de un tipo, escalera de color, 2+2+1.
\end{ejercicio}

\begin{ejercicio}
Moneda lanzada 10 veces: ¿cuántos resultados con 3 caras?
\end{ejercicio}

\begin{ejercicio}
Cadenas de 8 bits con a lo más 3 ceros.
\end{ejercicio}

\begin{ejercicio}
Comité de 4 republicanos, 3 demócratas, 2 independientes.
\end{ejercicio}

\begin{ejercicio}
Palabras clave de 4 caracteres.
\end{ejercicio}

\begin{ejercicio}
Ordenar 4 libros de matemáticas, 6 de física y 2 de química.
\end{ejercicio}

\begin{ejercicio}
Cadenas con letras ABCDEF bajo condiciones.
\end{ejercicio}

\begin{ejercicio}
Repartir 25 rosquillas entre 4 personas con restricciones.
\end{ejercicio}

\section{Relaciones de Recurrencia}

\begin{ejercicio}
$R_n$ = regiones por $n$ líneas. Calcular $R_1, R_2, R_3$ y relación de recurrencia.
\end{ejercicio}

\begin{ejercicio}
$S_n$ = cadenas de $n$ bits con $C(\alpha) \leq 2$.
\end{ejercicio}

\begin{ejercicio}
Resolver: $2a_n = 7a_{n-1} - 3a_{n-2} + 2^n$, etc.
\end{ejercicio}

\section{Funciones Generatrices}

\begin{ejercicio}
Asignar dos docenas de robots a 4 líneas de ensamble.
\end{ejercicio}

\begin{ejercicio}
300 sobres en paquetes de 25 entre 4 grupos.
\end{ejercicio}

\begin{ejercicio}
Distribuir $n$ objetos en $k$ cajas con al menos 1.
\end{ejercicio}

\begin{ejercicio}
Elegir 5 frutas entre manzanas, naranjas y peras.
\end{ejercicio}

\section{Análisis de Algoritmos}

\begin{ejercicio}
Notación $\Theta$ para $(n+1)(n+3)$ y $2\log n + 4n + 5n\log n$.
\end{ejercicio}

\begin{ejercicio}
Orden de los bucles anidados.
\end{ejercicio}

\begin{ejercicio}
Algoritmo recursivo máximo/mínimo: relación de recurrencia.
\end{ejercicio}

\begin{ejercicio}
Ordenamiento por inserción: peor caso.
\end{ejercicio}

\begin{ejercicio}
Evaluación de polinomios.
\end{ejercicio}

\begin{ejercicio}
Exponenciación rápida: complejidad $\Theta(\log n)$.
\end{ejercicio}

\section{Problemas Integradores}
\begin{ejercicio}
Demuestre que el número de formas de distribuir $n$ objetos idénticos en $k$ cajas distintas es $\binom{n+k-1}{k-1}$.
\end{ejercicio}

\begin{ejercicio}
Analice la complejidad del ordenamiento por inserción ($O(n^2)$).
\end{ejercicio}
